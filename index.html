const SHEET_NAME = 'Reviews';

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['id', 'name', 'reviewer', 'lat', 'lng', 'rating', 'frosting', 'gooeyness', 'price', 'notes', 'createdAt']);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function doGet() {
  const sheet = getOrCreateSheet();
  const rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  const headers = rows[0];
  const data = rows.slice(1).filter(r => r[1]).map(row => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    obj.lat = Number(obj.lat);
    obj.lng = Number(obj.lng);
    obj.rating = Number(obj.rating);
    obj.gooeyness = Number(obj.gooeyness);
    obj.createdAt = Number(obj.createdAt) || Date.now();
    return obj;
  });
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = getOrCreateSheet();
  const payload = JSON.parse(e.postData.contents);
  const id = payload.id || ('gs-' + new Date().getTime());
  const createdAt = payload.createdAt || new Date().getTime();
  sheet.appendRow([
    id,
    payload.name || '',
    payload.reviewer || '',
    Number(payload.lat),
    Number(payload.lng),
    Number(payload.rating) || 5,
    payload.frosting || 'Cream Cheese',
    Number(payload.gooeyness) || 4,
    payload.price || '',
    payload.notes || '',
    createdAt
  ]);
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', id: id })).setMimeType(ContentService.MimeType.JSON);
}
